1. Reading Accounts & Logs (Instant):

Make sure your Google Sheet sharing is set to Anyone with the link (Viewer). Tabs supported: Users (or Sheet1), Attendance, and Logs.

2. Saving New Accounts, Attendance & Logs Back to Google Sheet:

Paste your deployed Google Apps Script Web App URL below so registrations, check-ins, and system logs write directly to your spreadsheet.

In your Google Sheet, click Extensions → Apps Script, paste this complete code (supports both doGet reading and doPost saving), click Deploy → New deployment → Web app (Execute as: Me, Who has access: Anyone), and paste the /exec Web App URL above:

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  function readTab(names) {
    var sheet = null;
    for (var i = 0; i < names.length; i++) {
      sheet = ss.getSheetByName(names[i]);
      if (sheet) break;
    }
    if (!sheet) sheet = ss.getSheets()[0];
    if (!sheet || sheet.getLastRow() < 2) return [];
    var values = sheet.getDataRange().getValues();
    var headers = values[0].map(function(h) { return String(h).trim(); });
    var rows = [];
    for (var r = 1; r < values.length; r++) {
      var obj = {};
      for (var c = 0; c < headers.length; c++) {
        obj[headers[c]] = values[r][c];
      }
      rows.push(obj);
    }
    return rows;
  }
  var result = {
    ok: true,
    users: readTab(["Users", "Sheet1"]),
    attendance: readTab(["Attendance"]),
    logs: readTab(["Logs"])
  };
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var payload = JSON.parse(e.postData.contents);
  var action = payload.action;
  var data = payload.data || {};

  function getOrCreateSheet(name, headers) {
    var sheet = ss.getSheetByName(name);
    if (!sheet) {
      sheet = ss.insertSheet(name);
      sheet.appendRow(headers);
    } else if (sheet.getLastRow() === 0) {
      sheet.appendRow(headers);
    }
    return sheet;
  }

  if (action === "ping") {
    var logSheet = getOrCreateSheet("Logs", ["Date", "Time", "Activity", "User"]);
    logSheet.appendRow([data.date || new Date().toISOString().slice(0, 10), data.time || new Date().toLocaleTimeString(), data.activity || "Connection Test", data.user || "System"]);
    return ContentService.createTextOutput(JSON.stringify({ ok: true, message: "Saved test log to Google Sheet" })).setMimeType(ContentService.MimeType.JSON);
  }

  if (action === "upsertUser") {
    var sheet = getOrCreateSheet("Users", ["Name", "ID", "Grade", "Section", "Password", "Active", "Role"]);
    var rows = sheet.getDataRange().getValues();
    var updated = false;
    for (var i = 1; i < rows.length; i++) {
      if (String(rows[i][1]).trim().toLowerCase() === String(data.id).trim().toLowerCase()) {
        sheet.getRange(i + 1, 1, 1, 7).setValues([[data.name, data.id, data.grade, data.section, data.password, data.active !== false ? "TRUE" : "FALSE", data.role || "student"]]);
        updated = true;
        break;
      }
    }
    if (!updated) {
      sheet.appendRow([data.name, data.id, data.grade, data.section, data.password, data.active !== false ? "TRUE" : "FALSE", data.role || "student"]);
    }
  } else if (action === "deleteUser") {
    var uSheet = ss.getSheetByName("Users");
    if (uSheet) {
      var uRows = uSheet.getDataRange().getValues();
      for (var j = uRows.length - 1; j >= 1; j--) {
        if (String(uRows[j][1]).trim().toLowerCase() === String(data.id).trim().toLowerCase()) {
          uSheet.deleteRow(j + 1);
        }
      }
    }
  } else if (action === "addAttendance") {
    var aSheet = getOrCreateSheet("Attendance", ["Student", "StudentID", "Grade", "Section", "Date", "Time", "Status", "Remarks"]);
    aSheet.appendRow([data.student, data.studentID, data.grade, data.section, data.date, data.time, data.status, data.remarks]);
  } else if (action === "updateAttendanceStatus") {
    var attSheet = ss.getSheetByName("Attendance");
    if (attSheet) {
      var aRows = attSheet.getDataRange().getValues();
      for (var k = 1; k < aRows.length; k++) {
        if (String(aRows[k][1]).trim() === String(data.studentID).trim() && String(aRows[k][4]).trim() === String(data.date).trim()) {
          attSheet.getRange(k + 1, 7).setValue(data.status);
        }
      }
    }
  } else if (action === "addLog") {
    var lSheet = getOrCreateSheet("Logs", ["Date", "Time", "Activity", "User"]);
    lSheet.appendRow([data.date, data.time, data.activity, data.user]);
  }

  return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
}
